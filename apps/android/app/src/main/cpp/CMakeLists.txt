cmake_minimum_required(VERSION 3.22.1)
project(hydra_android LANGUAGES C CXX)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set -PllamaDir to verified llama.cpp b11429 sources.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(hydra_llama SHARED hydra_jni.cpp)
target_compile_features(hydra_llama PRIVATE cxx_std_17)
target_link_libraries(hydra_llama PRIVATE llama android log)
target_link_options(hydra_llama PRIVATE "-Wl,-z,max-page-size=16384")
